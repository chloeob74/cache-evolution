PROMISING STRATEGIES TO EXPLORE:
- SLRU (Segmented LRU): Two segments (probation/protected), promote on second access
- LRU-2: Track second-to-last access time, evict by that (not most recent)
- 2Q: Separate queues for first-time (A1) and repeat accesses (Am)
- ARC-lite: Adaptive balance between recency and frequency queues
- Frequency-aware: Use separate OrderedDicts per frequency level (freq=1, freq=2, freq>=3)

EXAMPLE - SLRU (Segmented LRU) with O(1) operations:

    def __init__(self, capacity_bytes):
        self.capacity = capacity_bytes
        self.protected_capacity = int(capacity_bytes * 0.6)  # 60% protected
        self.current_size = 0
        self.protected_size = 0
        self.hits = 0
        self.misses = 0
        # Two segments: probation (new items) and protected (accessed twice+)
        self.probation = OrderedDict()  # obj_id -> obj_size
        self.protected = OrderedDict()  # obj_id -> obj_size
    
    def access(self, obj_id, obj_size):
        # Check protected segment first (hot items)
        if obj_id in self.protected:
            self.protected.move_to_end(obj_id)
            self.hits += 1
            return True
        
        # Check probation segment
        if obj_id in self.probation:
            # Promote to protected on second access
            size = self.probation.pop(obj_id)
            self._add_to_protected(obj_id, size)
            self.hits += 1
            return True
        
        # Miss - add to probation
        self.misses += 1
        if obj_size > self.capacity:
            return False
        self._evict_until_fits(obj_size)
        self.probation[obj_id] = obj_size
        self.current_size += obj_size
        return False
    
    def _add_to_protected(self, obj_id, obj_size):
        # Make room in protected if needed by demoting to probation
        while self.protected_size + obj_size > self.protected_capacity and self.protected:
            demoted_id, demoted_size = self.protected.popitem(last=False)
            self.probation[demoted_id] = demoted_size  # demote to probation
        self.protected[obj_id] = obj_size
        self.protected_size += obj_size
    
    def _evict_until_fits(self, needed_size):
        # Evict from probation first (LRU order), then protected if needed
        while self.current_size + needed_size > self.capacity:
            if self.probation:
                evicted_id, evicted_size = self.probation.popitem(last=False)
            elif self.protected:
                evicted_id, evicted_size = self.protected.popitem(last=False)
                self.protected_size -= evicted_size
            else:
                break
            self.current_size -= evicted_size
